cmake_minimum_required(VERSION 3.18)
project(FPLUS VERSION 0.5.0 LANGUAGES C)

# ---------------------------------------------------------------------------- options
option(FPLUS_OPENMP "OpenMP parallel loops" ON)
option(FPLUS_BLAS "BLAS kernels through the CBLAS interface" ON)
set(FPLUS_BLAS_LIBRARIES "" CACHE STRING
    "BLAS link line overriding the search (e.g. -lopenblas, the Cray libsci library, or NVPL by full path)")
set(FPLUS_MARCH "native" CACHE STRING
    "-march value; empty for the compiler default, x86-64-v3 (or similar) for a build that must run on other CPUs")
option(FPLUS_PROFILE "phase timers in the supernodal kernel (global state: not thread-safe)" OFF)
option(FPLUS_BUILD_TESTS "the C tests (ctest)" ON)
option(FPLUS_BUILD_EXAMPLES "the C example program" ON)
option(FPLUS_BUILD_PYTHON "the CPython extension _sparse_lu_c (scikit-build-core turns this on)" OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "build type" FORCE)
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

# the flags of setup.py, so both builds give the same factors
set(FPLUS_C_FLAGS -O3 -fno-math-errno -fno-signed-zeros -DNDEBUG)
if(FPLUS_MARCH)
  list(APPEND FPLUS_C_FLAGS -march=${FPLUS_MARCH})
endif()

# ---------------------------------------------------------------------------- dependencies
# SuiteSparse COLAMD (required): its CMake package when available, else a header / library search
find_package(COLAMD CONFIG QUIET)
if(TARGET SuiteSparse::COLAMD)
  set(FPLUS_COLAMD SuiteSparse::COLAMD)
elseif(TARGET SuiteSparse::COLAMD_static)
  set(FPLUS_COLAMD SuiteSparse::COLAMD_static)
else()
  find_path(COLAMD_INCLUDE_DIR colamd.h PATH_SUFFIXES suitesparse)
  find_library(COLAMD_LIBRARY colamd)
  if(NOT COLAMD_INCLUDE_DIR OR NOT COLAMD_LIBRARY)
    message(FATAL_ERROR "SuiteSparse COLAMD not found (colamd.h / libcolamd): set CMAKE_PREFIX_PATH or COLAMD_INCLUDE_DIR / COLAMD_LIBRARY")
  endif()
  add_library(fplus_colamd INTERFACE)
  target_include_directories(fplus_colamd INTERFACE ${COLAMD_INCLUDE_DIR})
  target_link_libraries(fplus_colamd INTERFACE ${COLAMD_LIBRARY})
  set(FPLUS_COLAMD fplus_colamd)
endif()
message(STATUS "FPLUS: COLAMD ${FPLUS_COLAMD}")

if(FPLUS_BLAS)
  if(FPLUS_BLAS_LIBRARIES)
    separate_arguments(FPLUS_BLAS_LINK UNIX_COMMAND "${FPLUS_BLAS_LIBRARIES}")
  else()
    # a library that exports the cblas_* symbols: libcblas (conda, distributions), else OpenBLAS / the found BLAS
    # (a libcblas / OpenBLAS found first, e.g. in the CMAKE_PREFIX_PATH environment, is used alone: adding FindBLAS's
    # result too could mix two BLAS implementations and their threading runtimes in one process)
    find_library(FPLUS_CBLAS_LIBRARY NAMES cblas openblas)
    if(FPLUS_CBLAS_LIBRARY)
      set(FPLUS_BLAS_LINK ${FPLUS_CBLAS_LIBRARY})
    else()
      find_package(BLAS)
      set(FPLUS_BLAS_LINK ${BLAS_LIBRARIES})
    endif()
    if(NOT FPLUS_BLAS_LINK)
      message(FATAL_ERROR "no BLAS with the CBLAS interface found: set FPLUS_BLAS_LIBRARIES, or FPLUS_BLAS=OFF")
    endif()
  endif()
  message(STATUS "FPLUS: BLAS ${FPLUS_BLAS_LINK}")
endif()

if(FPLUS_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS C)
endif()

# ---------------------------------------------------------------------------- the core
set(FPLUS_SOURCES src/factor_row.c src/factor_sn.c src/equil.c src/solve.c src/order.c src/pipeline.c src/numeric.c src/solve_plan.c)
add_library(fplus_core OBJECT ${FPLUS_SOURCES})
set_target_properties(fplus_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(fplus_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> PRIVATE src)
target_compile_options(fplus_core PRIVATE ${FPLUS_C_FLAGS})
target_link_libraries(fplus_core PUBLIC ${FPLUS_COLAMD})
if(FPLUS_BLAS)
  target_compile_definitions(fplus_core PRIVATE HAVE_BLAS)
endif()
if(FPLUS_PROFILE)
  target_compile_definitions(fplus_core PRIVATE FPLUS_PROFILE)
endif()
if(FPLUS_OPENMP)
  target_link_libraries(fplus_core PUBLIC OpenMP::OpenMP_C)
endif()

set(FPLUS_LINK ${FPLUS_COLAMD} ${FPLUS_BLAS_LINK} ${CMAKE_DL_LIBS} m)
if(FPLUS_OPENMP)
  list(APPEND FPLUS_LINK OpenMP::OpenMP_C)
endif()

if(NOT FPLUS_BUILD_PYTHON)
  add_library(fplus SHARED $<TARGET_OBJECTS:fplus_core>)
  target_include_directories(fplus PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(fplus PRIVATE ${FPLUS_LINK})
  set_target_properties(fplus PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR})
  include(GNUInstallDirs)
  install(TARGETS fplus LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(FILES include/fplus.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

  if(FPLUS_BUILD_TESTS)
    enable_testing()
    add_executable(test_fplus tests/c/test_fplus.c)
    target_link_libraries(test_fplus PRIVATE fplus m)
    add_test(NAME fplus_c_api COMMAND test_fplus)
  endif()
  if(FPLUS_BUILD_EXAMPLES)
    add_executable(fplus_example examples/solve.c)
    target_link_libraries(fplus_example PRIVATE fplus m)
  endif()
endif()

# ---------------------------------------------------------------------------- the Python extension (scikit-build-core)
if(FPLUS_BUILD_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module NumPy)
  Python_add_library(_sparse_lu_c MODULE WITH_SOABI _sparse_lu_c.c $<TARGET_OBJECTS:fplus_core>)
  target_include_directories(_sparse_lu_c PRIVATE include)
  target_compile_options(_sparse_lu_c PRIVATE ${FPLUS_C_FLAGS})
  target_link_libraries(_sparse_lu_c PRIVATE Python::NumPy ${FPLUS_LINK})
  # the package FPLUS: the extension next to the Python layer (the repository root is the package in place)
  install(TARGETS _sparse_lu_c LIBRARY DESTINATION FPLUS)
  install(FILES __init__.py sparse_lu_full_pivot.py DESTINATION FPLUS)
endif()
